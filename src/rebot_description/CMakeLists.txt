cmake_minimum_required(VERSION 3.22)
project(rebot_description)
find_package(ament_cmake REQUIRED)
set(UPSTREAM ${CMAKE_CURRENT_SOURCE_DIR}/../external_dependencies/rebot_description)
if(NOT EXISTS ${UPSTREAM}/Rebot_Arm_description/RS/meshes)
  message(FATAL_ERROR "Initialize the pinned rebot_description submodule before building")
endif()
install(DIRECTORY ${UPSTREAM}/Rebot_Arm_description/RS/meshes DESTINATION share/${PROJECT_NAME})
install(DIRECTORY urdf DESTINATION share/${PROJECT_NAME})
install(FILES ${UPSTREAM}/LICENSE NOTICE.md DESTINATION share/${PROJECT_NAME})
ament_package()
